<script setup>
import { ref, computed, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'

// Estado para controlar qué tarjeta está en hover
const hoveredCard = ref(null)
const route = useRoute()
const router = useRouter()

// Estado para búsqueda global
const busquedaGlobal = ref('')
const buscandoGlobal = ref(false)
const resultadosGlobal = ref([])
const categoriaActiva = ref('')

// Variables para paginación de resultados de búsqueda global
const resultadosPaginaActual = ref(1)
const resultadosItemsPorPagina = ref(5)

// Variables para ordenamiento de resultados de búsqueda
const resultadosOrden = ref({
  columna: 'gestion',
  direccion: 'desc'
})

// Mapa de íconos por filtro
const iconosPorFiltro = ref({})

// Configuración de endpoints para gestión
const ENDPOINTS = {
  'resoluciones-camarales': {
    url: 'https://apisi.senado.gob.bo/page/resolucion-camarales',
    titulo: 'Resoluciones Camarales',
    icono: '',
    procesarRespuesta: (data) => {
      if (data.data && data.data.data) {
        return data.data.data.map(item => ({ ...item, _categoria: 'resoluciones-camarales' }))
      }
      return []
    },
    obtenerTotalPaginas: (data) => {
      return data.data?.last_page || 1
    }
  },
  'declaraciones-camarales': {
    url: 'https://apisi.senado.gob.bo/page/declaraciones-camarales',
    titulo: 'Declaraciones Camarales',
    icono: '',
    procesarRespuesta: (data) => {
      if (data.data && data.data.data) {
        return data.data.data.map(item => ({ ...item, _categoria: 'declaraciones-camarales' }))
      }
      return []
    },
    obtenerTotalPaginas: (data) => {
      return data.data?.last_page || 1
    }
  },
  'minutas-comunicacion': {
    url: 'https://apisi.senado.gob.bo/page/minutas-comunicacion',
    titulo: 'Minutas de Comunicación',
    icono: '',
    procesarRespuesta: (data) => {
      if (data.data && data.data.data) {
        return data.data.data.map(item => ({ ...item, _categoria: 'minutas-comunicacion' }))
      }
      return []
    },
    obtenerTotalPaginas: (data) => {
      return data.data?.last_page || 1
    }
  }
}

const props = defineProps({
  hash: {
    type: [String, Object],
    default: "gestion"
  },
  query: {
    type: Object,
    validator: (obj) => {
      return obj && typeof obj === 'object'
    }
  }
})

const emit = defineEmits(['update:query'])

// Datos de tabs desde la API externa
const tabsData = ref(null)
const cargandoTabs = ref(false)
const errorTabs = ref(null)

// Datos para la tabla
const todosLosDatos = ref([])
const loading = ref(false)
const error = ref(null)
const paginaActual = ref(1)
const itemsPorPagina = ref(5)
const terminoBusquedaLocal = ref('')
const busquedaLocalActiva = ref('')
const orden = ref({
  columna: 'titulo',
  direccion: 'desc'
})

// Funciones para extraer datos del título para ordenamiento en gestión
const extraerNumeroTitulo = (titulo) => {
  if (!titulo) return 0
  const match = titulo.match(/(\d+)/)
  return match ? parseInt(match[1], 10) : 0
}

const extraerAnioTitulo = (titulo) => {
  if (!titulo) return 0
  const match = titulo.match(/\/(\d{4})-\d{4}/)
  return match ? parseInt(match[1], 10) : 0
}

// Computed properties
const hashValido = computed(() => {
  return props.hash === 'permitido' || true
})

const filtroActual = computed(() => {
  return props.query?.filtro || route.query?.filtro
})

const filtroValido = computed(() => {
  return ENDPOINTS.hasOwnProperty(filtroActual.value)
})

const endpointConfig = computed(() => {
  if (!filtroValido.value) return null
  return ENDPOINTS[filtroActual.value]
})

const tituloSeccion = computed(() => {
  return endpointConfig.value?.titulo || 'Resoluciones Camarales'
})

// Icono activo para el título de la tabla
const iconoActivo = computed(() => {
  if (!filtroActual.value) return ''
  if (iconosPorFiltro.value[filtroActual.value]) {
    return iconosPorFiltro.value[filtroActual.value]
  }
  const link = linksGestion.value.find(l => {
    return l.path && l.path.includes(`filtro=${filtroActual.value}`)
  })
  return link?.icono || ''
})

// Datos de gestión desde la API de tabs
const areaGestion = computed(() => {
  return tabsData.value?.areas?.gestion || null
})

const linksGestion = computed(() => {
  return tabsData.value?.links?.gestion || []
})

// Computed para resultados ordenados correctamente
const resultadosOrdenados = computed(() => {
  if (!resultadosGlobal.value.length) return []
  
  return [...resultadosGlobal.value].sort((a, b) => {
    if (resultadosOrden.value.columna === 'gestion') {
      const gestionA = a.gestion || 0
      const gestionB = b.gestion || 0
      if (resultadosOrden.value.direccion === 'asc') {
        return gestionA - gestionB
      } else {
        return gestionB - gestionA
      }
    } else if (resultadosOrden.value.columna === 'titulo') {
      const anioA = extraerAnioTitulo(a.titulo)
      const anioB = extraerAnioTitulo(b.titulo)
      if (anioA !== anioB) {
        return resultadosOrden.value.direccion === 'asc' ? anioA - anioB : anioB - anioA
      }
      const numeroA = extraerNumeroTitulo(a.titulo)
      const numeroB = extraerNumeroTitulo(b.titulo)
      if (resultadosOrden.value.direccion === 'asc') {
        return numeroA - numeroB
      } else {
        return numeroB - numeroA
      }
    } else if (resultadosOrden.value.columna === 'asunto') {
      const asuntoA = a.asunto || ''

      if (resultadosOrden.value.direccion === 'asc') {
        return asuntoA.localeCompare(asuntoB)
      } else {
        return asuntoB.localeCompare(asuntoA)
      }
    } else if (resultadosOrden.value.columna === 'institucion') {
      const instA = a.institucion_text || ''
      const instB = b.institucion_text || ''
      if (resultadosOrden.value.direccion === 'asc') {
        return instA.localeCompare(instB)
      } else {
        return instB.localeCompare(instA)

// Función para cargar datos de un endpoint completo
const cargarEndpointCompleto = async (key, endpoint) => {
  try {
    const primeraPagina = await obtenerPaginaEndpoint(key, endpoint, 1)
    if (!primeraPagina) return []
    
    const datosPrimeraPagina = endpoint.procesarRespuesta(primeraPagina) || []
    const todosLosItems = [...datosPrimeraPagina]
    
    const totalPaginas = endpoint.obtenerTotalPaginas(primeraPagina)
    
    if (totalPaginas > 1) {
      const paginasRestantes = Array.from({ length: totalPaginas - 1 }, (_, i) => i + 2)
      const promesas = paginasRestantes.map(page => obtenerPaginaEndpoint(key, endpoint, page))
      const resultados = await Promise.all(promesas)
      
      resultados.forEach(res => {
        if (res) {
          const datosPagina = endpoint.procesarRespuesta(res) || []
          todosLosItems.push(...datosPagina)
        }
      })
    }
    
    return todosLosItems
  } catch (err) {
    console.error(`Error cargando endpoint ${key}:`, err)
    return []
  }
}

const obtenerPaginaEndpoint = async (key, endpoint, page) => {
  const url = `${endpoint.url}?page=${page}`
  try {
    const response = await fetch(url)
    return await response.json()
  } catch (err) {
    console.error(`Error cargando página ${page} de ${key}:`, err)
    return null
  }
}

        (_, i) => i + 2
      )
      
      const promesas = paginasRestantes.map(page => obtenerPagina(page))
      const resultados = await Promise.all(promesas)
      
      resultados.forEach(res => {
        if (res) {
          const datosPagina = endpointConfig.value.procesarRespuesta(res) || []

    console.error('Error:', err)
  } finally {
    loading.value = false
  }
}

const cargarTabsGestion = async () => {
  cargandoTabs.value = true
  errorTabs.value = null
  
  try {
    const response = await fetch('http://demoback.senado.gob.bo/api/tabs')
    const data = await response.json()
    
    if (data?.success && data?.data) {
      tabsData.value = data.data
      
      if (tabsData.value?.links?.gestion) {
        tabsData.value.links.gestion.forEach(link => {
          const filtroKey = link.path?.match(/filtro=([^&#]+)/)?.[1]
          if (filtroKey) {
            iconosPorFiltro.value[filtroKey] = link.icono
            if (ENDPOINTS[filtroKey]) {
              ENDPOINTS[filtroKey].icono = link.icono
            }
          }
        })
      }
    } else {
      throw new Error('Formato de respuesta inválido')
    }
  } catch (err) {
    errorTabs.value = 'Error al cargar opciones: ' + err.message
    console.error('Error cargando tabs:', err)
  } finally {
    cargandoTabs.value = false
  }
}

const allImages = ref(generateImages())
const totalImages = computed(() => allImages.value.length)

// ============================================
// PAGINACIÓN
// ============================================
const itemsPerPage = 20
const currentPage = ref(1)
const totalPages = computed(() => Math.ceil(totalImages.value / itemsPerPage))

const paginatedImages = computed(() => {
  const start = (currentPage.value - 1) * itemsPerPage
  const end = start + itemsPerPage
  return allImages.value.slice(start, end)
})

const featuredImages = computed(() => allImages.value.slice(0, 4))

const visiblePages = computed(() => {
  const maxVisible = 10
  const half = Math.floor(maxVisible / 2)
  let start = Math.max(1, currentPage.value - half)
  let end = Math.min(totalPages.value, start + maxVisible - 1)
  
  if (end - start + 1 < maxVisible) {
    start = Math.max(1, end - maxVisible + 1)
  }
  
  const pages = []
  for (let i = start; i <= end; i++) {
    pages.push(i)
  }
  return pages
})

const goToPage = (page) => {
  currentPage.value = page
  const modalContainer = document.querySelector('.fixed.inset-0.z-50 > .container')
  if (modalContainer) {
    modalContainer.scrollTop = 0
  }
}

const prevPage = () => {
  if (currentPage.value > 1) currentPage.value--
}

const nextPage = () => {
  if (currentPage.value < totalPages.value) currentPage.value++
}

// ============================================
// ESTADOS DE MODALES
// ============================================
const showFullGallery = ref(false)
const showImageModal = ref(false)
const currentIndex = ref(0)
const currentImage = ref(null)

// ============================================
// FUNCIONES DE MODALES
// ============================================

const openGallery = () => {
  currentPage.value = 1
  showFullGallery.value = true
  document.body.style.overflow = 'hidden'
}

const closeFullGallery = () => {
  showFullGallery.value = false
  document.body.style.overflow = ''
}

const openImageModal = (image, index) => {
  currentImage.value = image
  currentIndex.value = index
  showImageModal.value = true
  document.body.style.overflow = 'hidden'
}

const closeImageViewer = () => {
  showImageModal.value = false
  currentImage.value = null
  document.body.style.overflow = ''
}

const nextImage = () => {
  if (currentIndex.value < totalImages.value - 1) {
    currentIndex.value++
    currentImage.value = allImages.value[currentIndex.value]
  }
}

const prevImage = () => {
  if (currentIndex.value > 0) {
    currentIndex.value--

  }
}

const goToImage = (index) => {
  currentIndex.value = index
  currentImage.value = allImages.value[index]
}

const handleKeydown = (e) => {
  if (!showImageModal.value && !showFullGallery.value) return

    if (e.key === 'ArrowLeft') prevImage()
    else if (e.key === 'ArrowRight') nextImage()
    else if (e.key === 'Escape') closeImageViewer()
  } else if (showFullGallery.value && e.key === 'Escape') {
    closeFullGallery()
  }
}
