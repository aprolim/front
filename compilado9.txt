<script setup>
import { ref, reactive, onMounted, onBeforeUnmount } from 'vue'
import { navigateTo, useRoute, useRouter } from '#app'
import { socialLinks } from './data/config'
import GlobalModals from '@/components/GlobalModals.vue'

const route = useRoute()
const router = useRouter()

console.log('🏗️ [LAYOUT] alter8.vue - INICIALIZANDO LAYOUT')
console.log('📍 [LAYOUT] Ruta actual:', route.path)

const sidebarOpen = ref(false)
const mobileMenuState = reactive({
  institution: false,
  legislators: false,
  legislative: false,
  culture: false,
  participation: false
})

const goToHome = () => {
  console.log('🏠 [LAYOUT] Navegando al inicio')
  sidebarOpen.value = false
  navigateTo('/')
  setTimeout(() => {
    const snapContainer = document.querySelector('.snap-container')
    if (snapContainer) {
      snapContainer.scrollTop = 0
      snapContainer.scrollTo({ top: 0, behavior: 'smooth' })
    }
    window.scrollTo(0, 0)
  }, 100)
}

const toggleSidebar = () => {
  sidebarOpen.value = !sidebarOpen.value
  if (!sidebarOpen.value) {
    Object.keys(mobileMenuState).forEach(key => {
      mobileMenuState[key] = false
    })
  }
}

const toggleMobileMenu = (menu) => {
  mobileMenuState[menu] = !mobileMenuState[menu]
}

const onEscape = (e) => {
  if (e.key === 'Escape' && sidebarOpen.value) {
    sidebarOpen.value = false
  }
}

onMounted(() => {
  console.log('✅ [LAYOUT] alter8.vue - LAYOUT MONTADO')
  document.addEventListener('keydown', onEscape)
})

onBeforeUnmount(() => {
  console.log('🗑️ [LAYOUT] alter8.vue - LAYOUT DESMONTANDO')
  document.removeEventListener('keydown', onEscape)
})
</script>

let scrollObserver = null

const initScrollObserver = () => {
  scrollObserver = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          if (entry.target === importantNewsRef.value) {
            isImportantNewsVisible.value = true;
          } else if (entry.target === moreNewsRef.value) {
            isMoreNewsVisible.value = true;
          } else if (entry.target === senateRef.value) {
            isSenateVisible.value = true;
          } else if (entry.target === museumRef.value) {
            isMuseumVisible.value = true;
          }
          
          entry.target.classList.add('animate-in');
          scrollObserver.unobserve(entry.target);
        }
      });
    },
    {
      threshold: 0.15,
      rootMargin: '0px 0px -30px 0px'
    }
  );
  
  const sections = [
    importantNewsRef.value,
    moreNewsRef.value,
    senateRef.value,
    museumRef.value
  ];
  
  sections.forEach(section => {
    if (section) {
      scrollObserver.observe(section);
    }
  });
};

  initScrollListener();
  await nextTick();
  showSection.value = true;
  initScrollObserver();
